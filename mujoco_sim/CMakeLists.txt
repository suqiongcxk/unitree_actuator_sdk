cmake_minimum_required(VERSION 3.16)
project(creeper_mujoco_sim VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CREEPER_ENABLE_ONNX "Enable ONNX Runtime policy support" ON)
option(CREEPER_FETCH_MUJOCO "Fetch MuJoCo when it is not installed" OFF)

find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(mujoco CONFIG QUIET)
if(NOT mujoco_FOUND AND CREEPER_FETCH_MUJOCO)
  include(FetchContent)
  set(MUJOCO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(MUJOCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(mujoco URL https://github.com/google-deepmind/mujoco/archive/refs/tags/3.3.5.tar.gz)
  FetchContent_MakeAvailable(mujoco)
endif()
if(NOT TARGET mujoco::mujoco)
  find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h)
  find_library(MUJOCO_LIBRARY NAMES mujoco)
  if(NOT MUJOCO_INCLUDE_DIR OR NOT MUJOCO_LIBRARY)
    message(FATAL_ERROR "MuJoCo not found. Set mujoco_DIR/CMAKE_PREFIX_PATH or use -DCREEPER_FETCH_MUJOCO=ON")
  endif()
  add_library(mujoco::mujoco UNKNOWN IMPORTED)
  set_target_properties(mujoco::mujoco PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_INCLUDE_DIR}")
endif()

set(CREEPER_HAS_ONNX 0)
if(CREEPER_ENABLE_ONNX)
  find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATH_SUFFIXES onnxruntime/core/session)
  find_library(ONNXRUNTIME_LIBRARY onnxruntime)
  if(ONNXRUNTIME_INCLUDE_DIR AND ONNXRUNTIME_LIBRARY)
    set(CREEPER_HAS_ONNX 1)
  else()
    message(STATUS "ONNX Runtime not found: default-pose simulation remains available")
  endif()
endif()

add_library(creeper_sim_core
  src/mujoco_backend.cpp src/simulation_controller.cpp src/policy_observation.cpp
  src/onnx_policy.cpp src/motion_safety.cpp)
target_include_directories(creeper_sim_core PUBLIC include)
target_link_libraries(creeper_sim_core PUBLIC Eigen3::Eigen mujoco::mujoco)
target_compile_definitions(creeper_sim_core PUBLIC CREEPER_HAS_ONNX=${CREEPER_HAS_ONNX})
target_compile_options(creeper_sim_core PRIVATE -Wall -Wextra -Wpedantic)
if(CREEPER_HAS_ONNX)
  target_include_directories(creeper_sim_core PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
  target_link_libraries(creeper_sim_core PRIVATE "${ONNXRUNTIME_LIBRARY}")
endif()

add_executable(mujoco_robot_control src/mujoco_main.cpp)
target_link_libraries(mujoco_robot_control PRIVATE creeper_sim_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(creeper_unit_tests tests/test_core.cpp)
  target_link_libraries(creeper_unit_tests PRIVATE creeper_sim_core)
  add_test(NAME core_contracts COMMAND creeper_unit_tests ${CMAKE_CURRENT_SOURCE_DIR}/models/creeper/creeper.xml)
  add_test(NAME headless_standing_10s COMMAND mujoco_robot_control --model ${CMAKE_CURRENT_SOURCE_DIR}/models/creeper/creeper.xml --headless --duration 10 --no-policy --keyframe standing)
  set_tests_properties(headless_standing_10s PROPERTIES TIMEOUT 30 PASS_REGULAR_EXPRESSION "result=PASS")
endif()
